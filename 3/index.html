デプロイの儀式（チェックリスト）

作った画面をURLで開けるようにする手続きです。詳しくは本書第1章。ここには手順だけ置いておきます。

デプロイの儀式・更新・切り分け
■ デプロイの儀式（最初の1回）
1. エディタ右上の「デプロイ」→「新しいデプロイ」
2. 歯車（種類の選択）→「ウェブアプリ」
3. 「次のユーザーとして実行」＝ 自分
4. 「アクセスできるユーザー」＝ 開く人で決める
   （自分だけ→自分のみ／職員・児童→組織内／保護者・外部→全員）
5. 「デプロイ」→「アクセスを承認」→ 許可の一覧で「すべて選択」にチェック →「続行」（初回だけ。英語なら Select all → Continue）
6. 「完了」→ URLを控える

■ 直したとき（2回目から）
「デプロイ」→「デプロイを管理」→ 鉛筆 →「新バージョン」→「デプロイ」
※「新しいデプロイ」は押さない（URLが変わります）

■ 「開けません」と言われたら
1. シークレットウィンドウで開いてもらう（マルチアカウントの切り分け）
2. アクセスできるユーザーの設定を確かめる
3. 「新しいデプロイ」でURLが変わっていないか確かめる
合言葉は「最初の1回だけ『新しいデプロイ』、直したら『デプロイを管理』」。「新しいデプロイ」を押し直すとURLが変わって、配ったQRやブックマークが全部無効になります。

/** カレンダーからシートへ書き出します。▶実行を押すのはこの関数（2冊目第1章と同じ）。 */
function main() {
  let calendar = CalendarApp.getCalendarById(CALENDAR_ID);

  if (!calendar) {
    try {
      calendar = CalendarApp.subscribeToCalendar(CALENDAR_ID);
    } catch (e) {
      throw new Error(
        '練習用カレンダーを追加できませんでした。\n' +
        'コード掲載ページにある「練習用カレンダーを追加」のリンクを押してから、\n' +
        'もう一度▶実行してください。'
      );
    }
  }

  const week = nextWeek_();
  const events = calendar.getEvents(week.start, week.end);

  const rows = events.map(function (event) {
    const startedAt = event.getStartTime();
    return [
      ymd_(startedAt),
      weekdayOf_(startedAt),
      event.isAllDayEvent() ? '終日' : Utilities.formatDate(startedAt, 'Asia/Tokyo', 'H:mm'),
      event.getTitle()
    ];
  });

  const sheet = getOrCreateSheet_();
  if (sheet.getLastRow() > 1) {
    sheet.getRange(2, 1, sheet.getLastRow() - 1, HEADER.length).clearContent();
  }

  if (rows.length === 0) {
    SpreadsheetApp.getActive().toast(
      week.label + ' の予定は0件でした。カレンダーに予定がない週です。',
      '書き出しました', 10
    );
    return;
  }

  sheet.getRange(2, 1, rows.length, HEADER.length).setValues(rows);
  sheet.getRange(2, 1, rows.length, 1).setNumberFormat('@');
  SpreadsheetApp.getActive().toast(week.label + ' の予定を ' + rows.length + ' 件、書き出しました。', '完了', 6);
}

/**
 * ここが今回の追加です。デプロイしたURLが開かれると、この関数が呼ばれます。
 * やっていることは「週予定シートを読んで、スマホ向けの1枚のページにして返す」だけ。
 * 本体はあくまでシートのほうです。シートを直接直せば、画面も次に開いたときから変わります。
 */
function doGet() {
  const sheet = SpreadsheetApp.getActive().getSheetByName(SHEET_NAME);
  const lastRow = sheet ? sheet.getLastRow() : 0;
  const rows = lastRow > 1
    ? sheet.getRange(2, 1, lastRow - 1, HEADER.length).getValues()
    : [];

  // 日付ごとにまとめます
  const days = [];
  const byDate = {};
  rows.forEach(function (row) {
    const date = String(row[0]).trim();
    if (!date) return;
    if (!byDate[date]) {
      byDate[date] = { date: date, weekday: String(row[1]), items: [] };
      days.push(byDate[date]);
    }
    byDate[date].items.push({ time: String(row[2]), title: String(row[3]) });
  });

  let body = '';
  if (days.length === 0) {
    body = '<p class="empty">まだ予定が書き出されていません。<br>スプレッドシートで ▶実行（main）を押してください。</p>';
  } else {
    days.forEach(function (day) {
      body += '<section><h2>' + esc_(mdLabel_(day.date)) + '（' + esc_(day.weekday) + '）</h2><ul>';
      day.items.forEach(function (item) {
        body += '<li><span class="time">' + esc_(item.time) + '</span>' + esc_(item.title) + '</li>';
      });
      body += '</ul></section>';
    });
  }

  const html =
    '<style>' +
    'body{font-family:sans-serif;margin:0;background:#f6f7f9;color:#202124;}' +
    'header{background:#188038;color:#fff;padding:14px 16px;font-size:20px;font-weight:bold;}' +
    'main{padding:12px;max-width:640px;margin:0 auto;}' +
    'section{background:#fff;border-radius:10px;padding:4px 14px 10px;margin:0 0 12px;box-shadow:0 1px 3px rgba(0,0,0,.12);}' +
    'h2{font-size:17px;border-bottom:2px solid #188038;padding-bottom:6px;margin:10px 0 4px;}' +
    'ul{list-style:none;margin:0;padding:0;}' +
    'li{font-size:17px;padding:8px 2px;border-bottom:1px solid #eee;}' +
    'li:last-child{border-bottom:none;}' +
    '.time{display:inline-block;min-width:64px;color:#188038;font-weight:bold;}' +
    '.empty{font-size:16px;padding:24px 8px;text-align:center;color:#5f6368;}' +
    'footer{text-align:center;color:#9aa0a6;font-size:12px;padding:8px 0 20px;}' +
    '</style>' +
    '<header>' + esc_(APP_TITLE) + '</header>' +
    '<main>' + body + '</main>' +
    '<footer>このページは「' + esc_(SHEET_NAME) + '」シートの中身を表示しています</footer>';

  return HtmlService.createHtmlOutput(html)
    .setTitle(APP_TITLE)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1');

/** 「週予定」シートを、なければ見出しつきで作ります。 */
function getOrCreateSheet_() {
  const ss = SpreadsheetApp.getActive();
  ss.setSpreadsheetTimeZone('Asia/Tokyo'); // 日付がずれないように、先に日本時間にそろえます

  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) sheet = ss.insertSheet(SHEET_NAME, 0);

  const header = sheet.getRange(1, 1, 1, HEADER.length);
  if (String(header.getValues()[0][0]) !== HEADER[0]) {
    header.setValues([HEADER]).setFontWeight('bold').setBackground('#e8eaed');
    sheet.setFrozenRows(1);
    sheet.setColumnWidth(1, 110);
    sheet.setColumnWidth(2, 50);
    sheet.setColumnWidth(3, 90);
    sheet.setColumnWidth(4, 380);
  }
  return sheet;
}

/** 来週の月曜0時から、土曜0時までを返します（日本時間で組み立て）。 */
function nextWeek_() {
  const today = Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy-MM-dd').split('-').map(Number);
  const anchor = Date.UTC(today[0], today[1] - 1, today[2]);
  const day = new Date(anchor).getUTCDay();
  const untilMonday = ((8 - day) % 7) || 7;
  const monday = anchor + untilMonday * 24 * 60 * 60 * 1000;

  const start = new Date(monday - 9 * 60 * 60 * 1000);
  const end = new Date(start.getTime() + 5 * 24 * 60 * 60 * 1000);

  return {
    start: start,
    end: end,
    label: Utilities.formatDate(start, 'Asia/Tokyo', 'M月d日') + 'の週'
  };
}

/** 2026-09-07 の形の文字列にします。 */
function ymd_(date) {
  return Utilities.formatDate(date, 'Asia/Tokyo', 'yyyy-MM-dd');
}

/** 日本時間での曜日を1文字で返します。 */
function weekdayOf_(date) {
  const n = ymd_(date).split('-').map(Number);
  const anchor = new Date(Date.UTC(n[0], n[1] - 1, n[2]));
  return ['日', '月', '火', '水', '木', '金', '土'][anchor.getUTCDay()];
}

/** 「2026-09-07」を「9月7日」の形にします。読めない値はそのまま返します。 */
function mdLabel_(text) {
  const m = String(text).match(/^(\d{4})-(\d{1,2})-(\d{1,2})$/);
  return m ? Number(m[2]) + '月' + Number(m[3]) + '日' : String(text);
}

/** 画面に出す文字を、そのままの形で安全に表示するための変換です。 */
function esc_(text) {
  return String(text)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;');
}

  let calendar = CalendarApp.getCalendarById(CALENDAR_ID);

  if (!calendar) {
    try {
      calendar = CalendarApp.subscribeToCalendar(CALENDAR_ID);
    } catch (e) {
      throw new Error(
        '練習用カレンダーを追加できませんでした。\n' +
        'コード掲載ページにある「練習用カレンダーを追加」のリンクを押してから、\n' +
        'もう一度▶実行してください。'
      );
    }
  }

  const week = nextWeek_();
  const events = calendar.getEvents(week.start, week.end);

  const rows = events.map(function (event) {
    const startedAt = event.getStartTime();
    return [
      ymd_(startedAt),
      weekdayOf_(startedAt),
      event.isAllDayEvent() ? '終日' : Utilities.formatDate(startedAt, 'Asia/Tokyo', 'H:mm'),
      event.getTitle()
    ];
  });

  const sheet = getOrCreateSheet_();
  if (sheet.getLastRow() > 1) {
    sheet.getRange(2, 1, sheet.getLastRow() - 1, HEADER.length).clearContent();
  }

  if (rows.length === 0) {
    SpreadsheetApp.getActive().toast(
      week.label + ' の予定は0件でした。カレンダーに予定がない週です。',
      '書き出しました', 10
    );
    return;
  }

  sheet.getRange(2, 1, rows.length, HEADER.length).setValues(rows);
  sheet.getRange(2, 1, rows.length, 1).setNumberFormat('@');
  SpreadsheetApp.getActive().toast(week.label + ' の予定を ' + rows.length + ' 件、書き出しました。', '完了', 6);
}

    ? sheet.getRange(2, 1, lastRow - 1, HEADER.length).getValues()
    : [];

  const days = [];
  const byDate = {};
  rows.forEach(function (row) {

  });
  return { days: days };
}

/** 来週の月曜0時から、土曜0時までを返します（日本時間で組み立て）。 */
function nextWeek_() {
  const today = Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy-MM-dd').split('-').map(Number);
  const anchor = Date.UTC(today[0], today[1] - 1, today[2]);
  const day = new Date(anchor).getUTCDay();
  const untilMonday = ((8 - day) % 7) || 7;
  const monday = anchor + untilMonday * 24 * 60 * 60 * 1000;

  const start = new Date(monday - 9 * 60 * 60 * 1000);
  const end = new Date(start.getTime() + 5 * 24 * 60 * 60 * 1000);

  return {
    start: start,
    end: end,
    label: Utilities.formatDate(start, 'Asia/Tokyo', 'M月d日') + 'の週'
  };
}

/** 2026-09-07 の形の文字列にします。 */
function ymd_(date) {
  return Utilities.formatDate(date, 'Asia/Tokyo', 'yyyy-MM-dd');
}

/** 日本時間での曜日を1文字で返します。 */
function weekdayOf_(date) {
  const n = ymd_(date).split('-').map(Number);
  const anchor = new Date(Date.UTC(n[0], n[1] - 1, n[2]));
  return ['日', '月', '火', '水', '木', '金', '土'][anchor.getUTCDay()];
}

/** 「2026-09-07」を「9月7日」の形にします。読めない値はそのまま返します。 */
function mdLabel_(text) {
  const m = String(text).match(/^(\d{4})-(\d{1,2})-(\d{1,2})$/);
  return m ? Number(m[2]) + '月' + Number(m[3]) + '日' : String(text);
}

/** 所要（分）から、終了時刻と次の開始時刻を計算します。 */
function recalcTimes() {
  const sheet = ensureSheet_();
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) {
    SpreadsheetApp.getUi().alert('先に「進行表をつくる」を実行するか、1行目に開始時刻を書いてください。');
    return;
  }

  const range = sheet.getRange(2, 1, lastRow - 1, 3);
  const values = range.getValues();
  let cursor = parseTime_(values[0][0]);

  if (cursor === null) {
    SpreadsheetApp.getUi().alert(
      'いちばん上の開始時刻が読めません。「8:30」のように、半角の数字とコロンで書いてください。\n' +
      '（いま入っているもの: ' + values[0][0] + '）'
    );
    return;
  }

  for (let i = 0; i < values.length; i++) {
    const minutes = Number(values[i][1]);
    if (!minutes || minutes <= 0) {
      values[i][0] = (i === 0) ? formatTime_(cursor) : '';
      values[i][2] = '';
      continue;
    }
    values[i][0] = formatTime_(cursor);
    values[i][2] = formatTime_(cursor + minutes);
    cursor += minutes;
  }

  range.setNumberFormat('@');
  range.setValues(values);
  markNow();
}

/** いま進んでいる項目に「▶」と色をつけます（シート側の表示）。 */
function markNow() {
  const sheet = SpreadsheetApp.getActive().getSheetByName(SHEET_NAME);
  if (!sheet) return 0;

  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return 0;

  const values = sheet.getRange(2, 1, lastRow - 1, 3).getValues();
  const now = nowMinutes_();

  const marks = [];
  const colors = [];
  let hit = 0;

  for (let i = 0; i < values.length; i++) {
    const start = parseTime_(values[i][0]);
    const end = parseTime_(values[i][2]);

    if (start === null || end === null) {
      marks.push(['']);
      colors.push(['#ffffff']);
      continue;
    }
    if (now >= start && now < end) {
      marks.push(['▶ あと' + (end - now) + '分']);
      colors.push([NOW_COLOR]);
      hit = i + 2;
    } else if (now >= end) {
      marks.push(['済']);
      colors.push([DONE_COLOR]);
    } else {
      marks.push(['']);
      colors.push(['#ffffff']);
    }
  }

  sheet.getRange(2, 6, marks.length, 1).setValues(marks);
  sheet.getRange(2, 1, colors.length, HEADER.length).setBackgrounds(
    colors.map(function (c) { return new Array(HEADER.length).fill(c[0]); })
  );

  if (hit > 0) {
    try {
      sheet.setActiveSelection(sheet.getRange(hit, 1, 1, HEADER.length));
    } catch (e) {
      // 画面を開いていないときは、選択できなくても構いません
    }
  }
  return hit;
}

/** 画面から20秒ごとに呼ばれて、いまの進行の状態を返します。本体はシートです。 */
function getBoardState() {
  const sheet = SpreadsheetApp.getActive().getSheetByName(SHEET_NAME);
  const lastRow = sheet ? sheet.getLastRow() : 0;
  const state = { title: APP_TITLE, items: [], now: null, next: null, message: '' };
  if (lastRow < 2) return state;

  const values = sheet.getRange(2, 1, lastRow - 1, 5).getValues();
  const now = nowMinutes_();

  values.forEach(function (row) {
    const start = parseTime_(row[0]);
    const end = parseTime_(row[2]);
    const content = String(row[3]).trim();
    if (start === null || end === null || !content) return;

    let itemState = '';
    if (now >= start && now < end) itemState = 'now';
    else if (now >= end) itemState = 'done';

    const item = {
      start: formatTime_(start),
      end: formatTime_(end),
      content: content,
      state: itemState
    };
    state.items.push(item);

    if (itemState === 'now') {
      state.now = { content: content, left: end - now };

/** 「8:30」を、0時からの分数に直します。読めないときは null。 */
function parseTime_(text) {
  // 手で打った「8:30」をスプレッドシートが時刻データに変えてしまうことがあるので、
  // その場合は見た目どおりの時刻に読み直します
  if (text instanceof Date) {
    text = Utilities.formatDate(text, SpreadsheetApp.getActive().getSpreadsheetTimeZone(), 'H:mm');
  }
  const s = String(text).trim().replace('：', ':');
  const m = s.match(/^(\d{1,2}):(\d{1,2})$/);
  if (!m) return null;
  const hour = Number(m[1]);
  const minute = Number(m[2]);
  if (hour > 23 || minute > 59) return null;
  return hour * 60 + minute;
}

/** 分数を「8:30」の形に戻します。 */
function formatTime_(minutes) {
  const wrapped = ((minutes % 1440) + 1440) % 1440;
  const hour = Math.floor(wrapped / 60);
  const minute = wrapped % 60;
  return hour + ':' + (minute < 10 ? '0' + minute : minute);
}

/** いまの時刻を、日本時間で「0時からの分数」にします。 */
function nowMinutes_() {
  const text = Utilities.formatDate(new Date(), 'Asia/Tokyo', 'HH:mm');
  return parseTime_(text);
}

/**

 * すでに書いてある行事や備考は、聞いたうえで残せます。

function buildYear() {
  const ui = SpreadsheetApp.getUi();
  const thisYear = Number(Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy'));
  const thisMonth = Number(Utilities.formatDate(new Date(), 'Asia/Tokyo', 'MM'));
  const defaultYear = (thisMonth >= 4) ? thisYear : thisYear - 1;

  const answer = ui.prompt(
    '何年度の予定表を作りますか',
    '西暦4桁で入力してください（例: ' + defaultYear + '）。\n' +
    defaultYear + ' と入れると、' + defaultYear + '年4月1日から' + (defaultYear + 1) + '年3月31日までを作ります。',
    ui.ButtonSet.OK_CANCEL
  );
  if (answer.getSelectedButton() !== ui.Button.OK) return;

  const year = Number(String(answer.getResponseText()).trim());
  if (!year || year < 2000 || year > 2100) {
    ui.alert('西暦4桁の数字で入力してください。（入力されたもの: ' + answer.getResponseText() + '）');
    return;
  }

  const ss = SpreadsheetApp.getActive();
  ss.setSpreadsheetTimeZone('Asia/Tokyo'); // 日付がずれないように、先に日本時間にそろえます

  let sheet = ss.getSheetByName(SHEET_NAME);
  if (sheet && sheet.getLastRow() > 1) {
    const ok = ui.alert(
      '「' + SHEET_NAME + '」にはすでに中身があります',
      '作り直すと、行事や備考も消えます。続けますか。',
      ui.ButtonSet.OK_CANCEL
    );
    if (ok !== ui.Button.OK) return;
    sheet.clear();
    sheet.setConditionalFormatRules([]);
  }
  if (!sheet) sheet = ss.insertSheet(SHEET_NAME, 0);

  const holidays = fetchHolidays_(year);
  const rows = [];
  const holidayRows = [];

  let cursor = Date.UTC(year, 3, 1);          // 4月1日
  const last = Date.UTC(year + 1, 2, 31);     // 翌年3月31日

  while (cursor <= last) {
    const d = new Date(cursor);
    const key = Utilities.formatDate(d, 'UTC', 'yyyy-MM-dd');
    const weekday = ['日', '月', '火', '水', '木', '金', '土'][d.getUTCDay()];
    const holidayName = holidays[key] || '';
    const closed = (weekday === '土' || weekday === '日' || holidayName !== '');

    rows.push([
      new Date(cursor + 3 * 60 * 60 * 1000), // 日本時間の正午。どのタイムゾーンでも日付がずれません
      weekday,
      closed ? '／' : '◯',
      closed ? '／' : '◯',
      holidayName,
      ''
    ]);
    if (holidayName) holidayRows.push(rows.length + 1); // 見出しの分で +1

    cursor += 24 * 60 * 60 * 1000;
  }

  sheet.getRange(1, 1, 1, HEADER.length).setValues([HEADER])
    .setFontWeight('bold').setBackground('#e8eaed').setHorizontalAlignment('center');
  sheet.getRange(2, 1, rows.length, HEADER.length).setValues(rows);
  sheet.getRange(2, 1, rows.length, 1).setNumberFormat('yyyy/MM/dd');
  sheet.getRange(2, 2, rows.length, 3).setHorizontalAlignment('center');
  sheet.setFrozenRows(1);

  sheet.setColumnWidth(3, 50);
  sheet.setColumnWidth(4, 50);
  sheet.setColumnWidth(5, 220);
  sheet.setColumnWidth(6, 320);

  // 祝日の行だけ、先に色を塗っておきます
  holidayRows.forEach(function (row) {
    sheet.getRange(row, 1, 1, HEADER.length).setBackground(SUNDAY_COLOR);
  });

  applyColorRules_(sheet, rows.length);
  jumpToToday();

  ui.alert(rows.length + '日分の行を作りました。祝日は ' + Object.keys(holidays).length + ' 日、行事の列に名前が入っています。');
}

/**

function jumpToToday() {
  const sheet = SpreadsheetApp.getActive().getSheetByName(SHEET_NAME);
  if (!sheet) return;

  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return;

  const today = Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy-MM-dd');
  const dates = sheet.getRange(2, 1, lastRow - 1, 1).getValues();

  for (let i = 0; i < dates.length; i++) {
    const value = dates[i][0];
    if (!(value instanceof Date)) continue;
    if (Utilities.formatDate(value, 'Asia/Tokyo', 'yyyy-MM-dd') === today) {
      sheet.setActiveSelection(sheet.getRange(i + 2, 1, 1, HEADER.length));
      return;
    }
  }
  sheet.setActiveSelection(sheet.getRange(2, 1));
}

function refillSchoolDays() {
  const ui = SpreadsheetApp.getUi();
  const sheet = SpreadsheetApp.getActive().getSheetByName(SHEET_NAME);
  if (!sheet || sheet.getLastRow() < 2) {
    ui.alert('先に「1年分の日付をつくる」を実行してください。');
    return;
  }

  const lastRow = sheet.getLastRow();
  const dates = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
  const first = dates[0][0];
  if (!(first instanceof Date)) {
    ui.alert('日付の列（A列）に日付が入っていません。');
    return;
  }

  const year = Number(Utilities.formatDate(first, 'Asia/Tokyo', 'yyyy'));
  const month = Number(Utilities.formatDate(first, 'Asia/Tokyo', 'MM'));
  const holidays = fetchHolidays_(month >= 4 ? year : year - 1);

  const marks = dates.map(function (row) {
    const value = row[0];
    if (!(value instanceof Date)) return ['', ''];
    const key = Utilities.formatDate(value, 'Asia/Tokyo', 'yyyy-MM-dd');
    const n = key.split('-').map(Number);
    const day = new Date(Date.UTC(n[0], n[1] - 1, n[2])).getUTCDay(); // 0=日 6=土
    const closed = (day === 0 || day === 6 || holidays[key]);
    return closed ? ['／', '／'] : ['◯', '◯'];
  });

  sheet.getRange(2, 3, marks.length, 2).setValues(marks);
  ui.alert('出校・給食を入れ直しました。');
}

/** 画面から呼ばれて、今日の行と向こう1週間の行事を返します。本体はシートです。 */
function getTodayData() {
  const sheet = SpreadsheetApp.getActive().getSheetByName(SHEET_NAME);
  const lastRow = sheet ? sheet.getLastRow() : 0;
  if (lastRow < 2) return { ready: false };

  const values = sheet.getRange(2, 1, lastRow - 1, HEADER.length).getValues();
  const todayText = Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy-MM-dd');
  const result = { ready: true, today: { label: labelOf_(todayText, ''), school: false, lunch: false, event: '', note: '' }, week: [] };

  values.forEach(function (row) {
    const dateText = toDateText_(row[0]);
    if (!dateText) return;
    const diff = dayDiff_(todayText, dateText);
    if (diff === 0) {
      result.today = {
        label: labelOf_(dateText, String(row[1])),
        school: String(row[2]).trim() === '◯',
        lunch: String(row[3]).trim() === '◯',
        event: String(row[4]).trim(),
        note: String(row[5]).trim()
      };
    } else if (diff > 0 && diff <= 7 && String(row[4]).trim()) {
      result.week.push({ label: labelOf_(dateText, String(row[1])), event: String(row[4]).trim() });
    }
  });
  return result;
}

/** シートの日付セル（文字でもDateでも）を「2026-09-01」の形にします。 */
function toDateText_(value) {
  if (value instanceof Date) {
    return Utilities.formatDate(value, SpreadsheetApp.getActive().getSpreadsheetTimeZone(), 'yyyy-MM-dd');
  }
  const m = String(value).trim().match(/^(\d{4})[-\/](\d{1,2})[-\/](\d{1,2})/);
  if (!m) return '';
  return m[1] + '-' + ('0' + m[2]).slice(-2) + '-' + ('0' + m[3]).slice(-2);
}

/** 「2026-09-01」どうしの日数差を返します。 */
function dayDiff_(fromText, toText) {
  const a = fromText.split('-').map(Number);
  const b = toText.split('-').map(Number);
  return Math.round((Date.UTC(b[0], b[1] - 1, b[2]) - Date.UTC(a[0], a[1] - 1, a[2])) / (24 * 60 * 60 * 1000));
}

/** 「9月1日（月）」の形の表示を作ります。 */
function labelOf_(dateText, weekday) {
  const n = dateText.split('-').map(Number);
  const w = weekday || ['日', '月', '火', '水', '木', '金', '土'][new Date(Date.UTC(n[0], n[1] - 1, n[2])).getUTCDay()];
  return n[1] + '月' + n[2] + '日（' + w + '）';
}

/**

 * こうしておくと、日付を書き足しても色が勝手についてきます。
 */
function applyColorRules_(sheet, rowCount) {
  const range = sheet.getRange(2, 1, rowCount, HEADER.length);
  const rules = [
    SpreadsheetApp.newConditionalFormatRule()
      .whenFormulaSatisfied('=$A2=TODAY()')
      .setBackground(TODAY_COLOR)
      .setRanges([range]).build(),
    SpreadsheetApp.newConditionalFormatRule()
      .whenFormulaSatisfied('=WEEKDAY($A2)=7')
      .setBackground(SATURDAY_COLOR)
      .setRanges([range]).build(),
    SpreadsheetApp.newConditionalFormatRule()
      .whenFormulaSatisfied('=WEEKDAY($A2)=1')
      .setBackground(SUNDAY_COLOR)
      .setRanges([range]).build()
  ];
  sheet.setConditionalFormatRules(rules);
}

/** 日本の祝日を、日付をキーにした一覧にして返します。 */
function fetchHolidays_(year) {
  const map = {};
  const calendar = CalendarApp.getCalendarById(HOLIDAY_CALENDAR_ID);
  if (!calendar) {
    // 番人：祝日カレンダーが読めなくても、黙って止まらない
    SpreadsheetApp.getActive().toast('祝日カレンダーが読めませんでした。土日だけで判定します。', 'お知らせ', 8);
    return map;
  }

  const start = new Date(Date.UTC(year, 3, 1) - 9 * 60 * 60 * 1000);
  const end = new Date(Date.UTC(year + 1, 3, 1) - 9 * 60 * 60 * 1000);
  calendar.getEvents(start, end).forEach(function (event) {
    const key = Utilities.formatDate(event.getStartTime(), 'Asia/Tokyo', 'yyyy-MM-dd');
    map[key] = event.getTitle();
  });
  return map;
}

  let daysAdded = 0;
  while (daysAdded < 3) {

function reserve(row, name) {
  const childName = String(name || '').trim();
  if (!childName) {
    return { ok: false, message: 'お子さんの名前を入力してください。' };
  }

  const lock = LockService.getScriptLock();
  lock.waitLock(10 * 1000);
  try {
    const sheet = SpreadsheetApp.getActive().getSheetByName(SHEET_NAME);
    const rowNumber = Number(row);
    if (!sheet || !rowNumber || rowNumber < 2 || rowNumber > sheet.getLastRow()) {
      return { ok: false, message: 'その枠が見つかりませんでした。ページを開き直してください。' };
    }

    const range = sheet.getRange(rowNumber, 1, 1, HEADER.length);
    const slot = range.getValues()[0];
    if (String(slot[3]).trim() !== '') {
      return { ok: false, message: 'その枠は、たったいま埋まりました。ほかの枠を選んでください。' };
    }

    sheet.getRange(rowNumber, 4).setValue(childName);
    return {
      ok: true,
      message: slot[0] + '（' + slot[1] + '）' + slot[2] + ' で予約しました。お待ちしています。'
    };
  } finally {
    lock.releaseLock();
  }
}

        taken.textContent = slot.time + ' 予約済み';
        list.appendChild(taken);
      } else {
        const button = document.createElement('button');
        button.className = 'slot';
        button.textContent = slot.time;
        button.onclick = function () { book(slot); };
        list.appendChild(button);
      }
    });
  }

  function book(slot) {
    const name = document.getElementById('childName').value.trim();
    if (!name) {
      showMessage(false, '先に、お子さんの名前を入力してください。');
      return;
    }
    if (!confirm(mdLabel(slot.date) + '（' + slot.weekday + '）' + slot.time + ' を予約します。よろしいですか？')) return;

    google.script.run
      .withSuccessHandler(function (result) {
        showMessage(result.ok, result.message);
        loadSlots(); // 最新の空き状況に描き直します
      })
      .withFailureHandler(function (e) { showMessage(false, '予約に失敗しました。もう一度お試しください。'); })
      .reserve(slot.row, name);
  }

  function showMessage(ok, text) {
    const box = document.getElementById('message');
    box.className = ok ? 'ok' : 'ng';
    box.textContent = text;
    window.scrollTo(0, 0);
  }

  function mdLabel(text) {
    const m = String(text).match(/^(\d{4})-(\d{1,2})-(\d{1,2})$/);
    return m ? Number(m[2]) + '月' + Number(m[3]) + '日' : String(text);
  }
